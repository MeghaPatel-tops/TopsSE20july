CREATE DATABASE ecom20july;
use ecom20july;

CREATE TABLE category(cid int PRIMARY KEY AUTO_INCREMENT,cname varchar(30));

DESC category;

CREATE TABLE product (pid int PRIMARY KEY AUTO_INCREMENT,pname varchar(40),price float,catid int ,description varchar(100),FOREIGN KEY(catid) REFERENCES category(cid),qty int);


create TABLE users(userid int PRIMARY KEY AUTO_INCREMENT,username varchar(20),email varchar(50), contact varchar(40));
